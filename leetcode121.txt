class Solution {
    public int maxProfit(int[] prices) {
        int max=prices[0],min=prices[0];
        int profit=max-min;
        for(int i=1;i<prices.length;i++){
             if(prices[i]<min){
                min=prices[i];
                max=prices[i];
                profit=max-min>profit?max-min:profit;
             }
             else{
                if(prices[i]>max)
                max=prices[i];
                profit=max-min>profit?max-min:profit;
             }
        }
        return profit;
    }
}
